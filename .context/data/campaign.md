  duration: "Video duration in seconds"

    - age_range: "18-35"
    - interests: ["list"]

  cta: "Call to action"
  tone: "professional | casual | dramatic | inspiring"

    colors: ["#hex"]
    voice: "Brand voice description"
    constraints: ["list"]
